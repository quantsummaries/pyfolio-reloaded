  58def timer(msg_body, previous_time):
  59    current_time = time()
  60    run_time = current_time - previous_time
  61    message = "\nFinished " + msg_body + " (required {:.2f} seconds)."
  62    print(message.format(run_time))
  63
  64    return current_time

  67def create_full_tear_sheet(
  68    returns,
  69    positions=None,
  70    transactions=None,
  71    market_data=None,
  72    benchmark_rets=None,
  73    slippage=None,
  74    live_start_date=None,
  75    sector_mappings=None,
  76    round_trips=False,
  77    estimate_intraday="infer",
  78    hide_positions=False,
  79    cone_std=(1.0, 1.5, 2.0),
  80    bootstrap=False,
  81    unadjusted_returns=None,
  82    turnover_denom="AGB",
  83    set_context=True,
  84    factor_returns=None,
  85    factor_loadings=None,
  86    pos_in_dollars=True,
  87    header_rows=None,
  88    factor_partitions=FACTOR_PARTITIONS,
  89):
  90    """
  91    Generate a number of tear sheets that are useful
  92    for analyzing a strategy's performance.
  93
  94    - Fetches benchmarks if needed.
  95    - Creates tear sheets for returns, and significant events.
  96        If possible, also creates tear sheets for position analysis
  97        and transaction analysis.
  98
  99    Parameters
 100    ----------
 101    returns : pd.Series
 102        Daily returns of the strategy, noncumulative.
 103        Time series with decimal returns.
 104
 105        Example::
 106
 107            2015-07-16    -0.012143
 108            2015-07-17    0.045350
 109            2015-07-20    0.030957
 110            2015-07-21    0.004902
 111    positions : pd.DataFrame, optional
 112        Daily net position values.
 113        Time series of dollar amount invested in each position and cash.
 114        Days where stocks are not held can be represented by 0 or NaN.
 115        Non-working capital is labelled 'cash'
 116
 117        Example::
 118
 119            index         'AAPL'         'MSFT'          cash
 120            2004-01-09    13939.3800     -14012.9930     711.5585
 121            2004-01-12    14492.6300     -14624.8700     27.1821
 122            2004-01-13    -13853.2800    13653.6400      -43.6375
 123    transactions : pd.DataFrame, optional
 124        Executed trade volumes and fill prices.
 125
 126        - One row per trade.
 127        - Trades on different names that occur at the
 128          same time will have identical indicies.
 129
 130        Example::
 131
 132            index                  amount   price    symbol
 133            2004-01-09 12:18:01    483      324.12   'AAPL'
 134            2004-01-09 12:18:01    122      83.10    'MSFT'
 135            2004-01-13 14:12:23    -75      340.43   'AAPL'
 136    market_data : pd.DataFrame, optional
 137        Daily market_data
 138        - DataFrame has a multi-index index, one level is dates and another is
 139        market_data contains volume & price, equities as columns
 140    slippage : int/float, optional
 141        Basis points of slippage to apply to returns before generating
 142        tearsheet stats and plots.
 143        If a value is provided, slippage parameter sweep
 144        plots will be generated from the unadjusted returns.
 145        Transactions and positions must also be passed.
 146        - See txn.adjust_returns_for_slippage for more details.
 147    live_start_date : datetime, optional
 148        The point in time when the strategy began live trading,
 149        after its backtest period. This datetime should be normalized.
 150    hide_positions : bool, optional
 151        If True, will not output any symbol names.
 152    round_trips: boolean, optional
 153        If True, causes the generation of a round trip tear sheet.
 154    sector_mappings : dict or pd.Series, optional
 155        Security identifier to sector mapping.
 156        Security ids as keys, sectors as values.
 157    estimate_intraday: boolean or str, optional
 158        Instead of using the end-of-day positions, use the point in the day
 159        where we have the most $ invested. This will adjust positions to
 160        better approximate and represent how an intraday strategy behaves.
 161        By default, this is 'infer', and an attempt will be made to detect
 162        an intraday strategy. Specifying this value will prevent detection.
 163    cone_std : float, or tuple, optional
 164        If float, The standard deviation to use for the cone plots.
 165        If tuple, Tuple of standard deviation values to use for the cone plots
 166        The cone is a normal distribution with this standard deviation
 167        centered around a linear regression.
 168    bootstrap : boolean (optional)
 169        Whether to perform bootstrap analysis for the performance
 170        metrics. Takes a few minutes longer.
 171    turnover_denom : str
 172        Either AGB or portfolio_value, default AGB.
 173        - See full explanation in txn.get_turnover.
 174    factor_returns : pd.Dataframe, optional
 175        Returns by factor, with date as index and factors as columns
 176    factor_loadings : pd.Dataframe, optional
 177        Factor loadings for all days in the date range, with date and
 178        ticker as index, and factors as columns.
 179    pos_in_dollars : boolean, optional
 180        indicates whether positions is in dollars
 181    header_rows : dict or OrderedDict, optional
 182        Extra rows to display at the top of the perf stats table.
 183    set_context : boolean, optional
 184        If True, set default plotting style context.
 185        See plotting.plotting_context().
 186    factor_partitions : dict, optional
 187        dict specifying how factors should be separated in perf attrib
 188        factor returns and risk exposures plots
 189        - See create_perf_attrib_tear_sheet().
 190    """
 191
 192    if (
 193        (unadjusted_returns is None)
 194        and (slippage is not None)
 195        and (transactions is not None)
 196    ):
 197        unadjusted_returns = returns.copy()
 198        returns = txn.adjust_returns_for_slippage(
 199            returns, positions, transactions, slippage
 200        )
 201
 202    positions = utils.check_intraday(
 203        estimate_intraday, returns, positions, transactions
 204    )
 205
 206    create_returns_tear_sheet(
 207        returns,
 208        positions=positions,
 209        transactions=transactions,
 210        live_start_date=live_start_date,
 211        cone_std=cone_std,
 212        benchmark_rets=benchmark_rets,
 213        bootstrap=bootstrap,
 214        turnover_denom=turnover_denom,
 215        header_rows=header_rows,
 216        set_context=set_context,
 217    )
 218
 219    create_interesting_times_tear_sheet(
 220        returns, benchmark_rets=benchmark_rets, set_context=set_context
 221    )
 222
 223    if positions is not None:
 224        create_position_tear_sheet(
 225            returns,
 226            positions,
 227            hide_positions=hide_positions,
 228            set_context=set_context,
 229            sector_mappings=sector_mappings,
 230            estimate_intraday=False,
 231        )
 232
 233        if transactions is not None:
 234            create_txn_tear_sheet(
 235                returns,
 236                positions,
 237                transactions,
 238                unadjusted_returns=unadjusted_returns,
 239                estimate_intraday=False,
 240                set_context=set_context,
 241            )
 242            if round_trips:
 243                create_round_trip_tear_sheet(
 244                    returns=returns,
 245                    positions=positions,
 246                    transactions=transactions,
 247                    sector_mappings=sector_mappings,
 248                    estimate_intraday=False,
 249                )
 250
 251            if market_data is not None:
 252                create_capacity_tear_sheet(
 253                    returns,
 254                    positions,
 255                    transactions,
 256                    market_data,
 257                    liquidation_daily_vol_limit=0.2,
 258                    last_n_days=125,
 259                    estimate_intraday=False,
 260                )
 261
 262        if factor_returns is not None and factor_loadings is not None:
 263            create_perf_attrib_tear_sheet(
 264                returns,
 265                positions,
 266                factor_returns,
 267                factor_loadings,
 268                transactions,
 269                pos_in_dollars=pos_in_dollars,
 270                factor_partitions=factor_partitions,
 271            )

 274@plotting.customize
 275def create_simple_tear_sheet(
 276    returns,
 277    positions=None,
 278    transactions=None,
 279    benchmark_rets=None,
 280    slippage=None,
 281    estimate_intraday="infer",
 282    live_start_date=None,
 283    turnover_denom="AGB",
 284    header_rows=None,
 285):
 286    """
 287    Simpler version of create_full_tear_sheet; generates summary performance
 288    statistics and important plots as a single image.
 289
 290    - Plots: cumulative returns, rolling beta, rolling Sharpe, underwater,
 291        exposure, top 10 holdings, total holdings, long/short holdings,
 292        daily turnover, transaction time distribution.
 293    - Never accept market_data input (market_data = None)
 294    - Never accept sector_mappings input (sector_mappings = None)
 295    - Never perform bootstrap analysis (bootstrap = False)
 296    - Never hide posistions on top 10 holdings plot (hide_positions = False)
 297    - Always use default cone_std (cone_std = (1.0, 1.5, 2.0))
 298
 299    Parameters
 300    ----------
 301    returns : pd.Series
 302        Daily returns of the strategy, noncumulative.
 303        Time series with decimal returns.
 304
 305        Example::
 306
 307            2015-07-16    -0.012143
 308            2015-07-17    0.045350
 309            2015-07-20    0.030957
 310            2015-07-21    0.004902
 311    positions : pd.DataFrame, optional
 312        Daily net position values.
 313        Time series of dollar amount invested in each position and cash.
 314        Days where stocks are not held can be represented by 0 or NaN.
 315        Non-working capital is labelled 'cash'
 316
 317        Example::
 318
 319            index         'AAPL'         'MSFT'          cash
 320            2004-01-09    13939.3800     -14012.9930     711.5585
 321            2004-01-12    14492.6300     -14624.8700     27.1821
 322            2004-01-13    -13853.2800    13653.6400      -43.6375
 323    transactions : pd.DataFrame, optional
 324        Executed trade volumes and fill prices.
 325
 326        - One row per trade.
 327        - Trades on different names that occur at the
 328          same time will have identical indicies.
 329
 330        Example::
 331
 332            index                  amount   price    symbol
 333            2004-01-09 12:18:01    483      324.12   'AAPL'
 334            2004-01-09 12:18:01    122      83.10    'MSFT'
 335            2004-01-13 14:12:23    -75      340.43   'AAPL'
 336    benchmark_rets : pd.Series, optional
 337        Daily returns of the benchmark, noncumulative.
 338    slippage : int/float, optional
 339        Basis points of slippage to apply to returns before generating
 340        tearsheet stats and plots.
 341        If a value is provided, slippage parameter sweep
 342        plots will be generated from the unadjusted returns.
 343        Transactions and positions must also be passed.
 344        - See txn.adjust_returns_for_slippage for more details.
 345    live_start_date : datetime, optional
 346        The point in time when the strategy began live trading,
 347        after its backtest period. This datetime should be normalized.
 348    turnover_denom : str, optional
 349        Either AGB or portfolio_value, default AGB.
 350        - See full explanation in txn.get_turnover.
 351    header_rows : dict or OrderedDict, optional
 352        Extra rows to display at the top of the perf stats table.
 353    set_context : boolean, optional
 354        If True, set default plotting style context.
 355    """
 356
 357    positions = utils.check_intraday(
 358        estimate_intraday, returns, positions, transactions
 359    )
 360
 361    if (slippage is not None) and (transactions is not None):
 362        returns = txn.adjust_returns_for_slippage(
 363            returns, positions, transactions, slippage
 364        )
 365
 366    always_sections = 4
 367    positions_sections = 4 if positions is not None else 0
 368    transactions_sections = 2 if transactions is not None else 0
 369    live_sections = 1 if live_start_date is not None else 0
 370    benchmark_sections = 1 if benchmark_rets is not None else 0
 371
 372    vertical_sections = sum(
 373        [
 374            always_sections,
 375            positions_sections,
 376            transactions_sections,
 377            live_sections,
 378            benchmark_sections,
 379        ]
 380    )
 381
 382    if live_start_date is not None:
 383        live_start_date = ep.utils.get_utc_timestamp(live_start_date)
 384
 385    plotting.show_perf_stats(
 386        returns,
 387        benchmark_rets,
 388        positions=positions,
 389        transactions=transactions,
 390        turnover_denom=turnover_denom,
 391        live_start_date=live_start_date,
 392        header_rows=header_rows,
 393    )
 394
 395    fig = plt.figure(figsize=(14, vertical_sections * 6))
 396    gs = gridspec.GridSpec(vertical_sections, 3, wspace=0.5, hspace=0.5)
 397
 398    ax_rolling_returns = plt.subplot(gs[:2, :])
 399    i = 2
 400    if benchmark_rets is not None:
 401        ax_rolling_beta = plt.subplot(gs[i, :], sharex=ax_rolling_returns)
 402        i += 1
 403    ax_rolling_sharpe = plt.subplot(gs[i, :], sharex=ax_rolling_returns)
 404    i += 1
 405    ax_underwater = plt.subplot(gs[i, :], sharex=ax_rolling_returns)
 406    i += 1
 407
 408    plotting.plot_rolling_returns(
 409        returns,
 410        factor_returns=benchmark_rets,
 411        live_start_date=live_start_date,
 412        cone_std=(1.0, 1.5, 2.0),
 413        ax=ax_rolling_returns,
 414    )
 415    ax_rolling_returns.set_title("Cumulative returns")
 416
 417    if benchmark_rets is not None:
 418        plotting.plot_rolling_beta(returns, benchmark_rets, ax=ax_rolling_beta)
 419
 420    plotting.plot_rolling_sharpe(returns, ax=ax_rolling_sharpe)
 421
 422    plotting.plot_drawdown_underwater(returns, ax=ax_underwater)
 423
 424    if positions is not None:
 425        # Plot simple positions tear sheet
 426        ax_exposures = plt.subplot(gs[i, :])
 427        i += 1
 428        ax_top_positions = plt.subplot(gs[i, :], sharex=ax_exposures)
 429        i += 1
 430        ax_holdings = plt.subplot(gs[i, :], sharex=ax_exposures)
 431        i += 1
 432        ax_long_short_holdings = plt.subplot(gs[i, :])
 433        i += 1
 434
 435        positions_alloc = pos.get_percent_alloc(positions)
 436
 437        plotting.plot_exposures(returns, positions, ax=ax_exposures)
 438
 439        plotting.show_and_plot_top_positions(
 440            returns,
 441            positions_alloc,
 442            show_and_plot=0,
 443            hide_positions=False,
 444            ax=ax_top_positions,
 445        )
 446
 447        plotting.plot_holdings(returns, positions_alloc, ax=ax_holdings)
 448
 449        plotting.plot_long_short_holdings(
 450            returns, positions_alloc, ax=ax_long_short_holdings
 451        )
 452
 453        if transactions is not None:
 454            # Plot simple transactions tear sheet
 455            ax_turnover = plt.subplot(gs[i, :])
 456            i += 1
 457            ax_txn_timings = plt.subplot(gs[i, :])
 458            i += 1
 459
 460            plotting.plot_turnover(
 461                returns,
 462                transactions,
 463                positions,
 464                turnover_denom=turnover_denom,
 465                ax=ax_turnover,
 466            )
 467
 468            plotting.plot_txn_time_hist(transactions, ax=ax_txn_timings)
 469
 470    for ax in fig.axes:
 471        ax.tick_params(
 472            axis="x",
 473            which="major",
 474            bottom=True,
 475            top=False,
 476            labelbottom=True,
 477        )

 480@plotting.customize
 481def create_returns_tear_sheet(
 482    returns,
 483    positions=None,
 484    transactions=None,
 485    live_start_date=None,
 486    cone_std=(1.0, 1.5, 2.0),
 487    benchmark_rets=None,
 488    bootstrap=False,
 489    turnover_denom="AGB",
 490    header_rows=None,
 491    return_fig=False,
 492):
 493    """
 494    Generate a number of plots for analyzing a strategy's returns.
 495
 496    - Fetches benchmarks, then creates the plots on a single figure.
 497    - Plots: rolling returns (with cone), rolling beta, rolling sharpe,
 498        rolling Fama-French risk factors, drawdowns, underwater plot, monthly
 499        and annual return plots, daily similarity plots,
 500        and return quantile box plot.
 501    - Will also print the start and end dates of the strategy,
 502        performance statistics, drawdown periods, and the return range.
 503
 504    Parameters
 505    ----------
 506    returns : pd.Series
 507        Daily returns of the strategy, noncumulative.
 508        See full explanation in create_full_tear_sheet.
 509    positions : pd.DataFrame, optional
 510        Daily net position values.
 511        See full explanation in create_full_tear_sheet.
 512    transactions : pd.DataFrame, optional
 513        Executed trade volumes and fill prices.
 514        - See full explanation in create_full_tear_sheet.
 515    live_start_date : datetime, optional
 516        The point in time when the strategy began live trading,
 517        after its backtest period.
 518    cone_std : float, or tuple, optional
 519        If float, The standard deviation to use for the cone plots.
 520        If tuple, Tuple of standard deviation values to use for the cone plots
 521        The cone is a normal distribution with this standard deviation
 522        centered around a linear regression.
 523    benchmark_rets : pd.Series, optional
 524        Daily noncumulative returns of the benchmark.
 525        This is in the same style as returns.
 526    bootstrap : boolean, optional
 527        Whether to perform bootstrap analysis for the performance
 528        metrics. Takes a few minutes longer.
 529    turnover_denom : str, optional
 530        Either AGB or portfolio_value, default AGB.
 531        - See full explanation in txn.get_turnover.
 532    header_rows : dict or OrderedDict, optional
 533        Extra rows to display at the top of the perf stats table.
 534    return_fig : boolean, optional
 535        If True, returns the figure that was plotted on.
 536    """
 537
 538    if benchmark_rets is not None:
 539        returns = utils.clip_returns_to_benchmark(returns, benchmark_rets)
 540
 541    if returns.empty:
 542        raise ValueError("returns must contain at least one observation")
 543
 544    plotting.show_perf_stats(
 545        returns,
 546        benchmark_rets,
 547        positions=positions,
 548        transactions=transactions,
 549        turnover_denom=turnover_denom,
 550        bootstrap=bootstrap,
 551        live_start_date=live_start_date,
 552        header_rows=header_rows,
 553    )
 554
 555    plotting.show_worst_drawdown_periods(returns)
 556
 557    vertical_sections = 11
 558
 559    if live_start_date is not None:
 560        vertical_sections += 1
 561        live_start_date = ep.utils.get_utc_timestamp(live_start_date)
 562
 563    if benchmark_rets is not None:
 564        vertical_sections += 1
 565
 566    if bootstrap:
 567        vertical_sections += 1
 568
 569    fig = plt.figure(figsize=(14, vertical_sections * 6))
 570    gs = gridspec.GridSpec(vertical_sections, 3, wspace=0.5, hspace=0.5)
 571    ax_rolling_returns = plt.subplot(gs[:2, :])
 572
 573    i = 2
 574    ax_rolling_returns_vol_match = plt.subplot(gs[i, :], sharex=ax_rolling_returns)
 575    i += 1
 576    ax_rolling_returns_log = plt.subplot(gs[i, :], sharex=ax_rolling_returns)
 577    i += 1
 578    ax_returns = plt.subplot(gs[i, :], sharex=ax_rolling_returns)
 579    i += 1
 580    if benchmark_rets is not None:
 581        ax_rolling_beta = plt.subplot(gs[i, :], sharex=ax_rolling_returns)
 582        i += 1
 583    ax_rolling_volatility = plt.subplot(gs[i, :], sharex=ax_rolling_returns)
 584    i += 1
 585    ax_rolling_sharpe = plt.subplot(gs[i, :], sharex=ax_rolling_returns)
 586    i += 1
 587    ax_drawdown = plt.subplot(gs[i, :], sharex=ax_rolling_returns)
 588    i += 1
 589    ax_underwater = plt.subplot(gs[i, :], sharex=ax_rolling_returns)
 590    i += 1
 591    ax_monthly_heatmap = plt.subplot(gs[i, 0])
 592    ax_annual_returns = plt.subplot(gs[i, 1])
 593    ax_monthly_dist = plt.subplot(gs[i, 2])
 594    i += 1
 595    ax_return_quantiles = plt.subplot(gs[i, :])
 596    i += 1
 597
 598    plotting.plot_rolling_returns(
 599        returns,
 600        factor_returns=benchmark_rets,
 601        live_start_date=live_start_date,
 602        cone_std=cone_std,
 603        ax=ax_rolling_returns,
 604    )
 605    ax_rolling_returns.set_title("Cumulative returns")
 606
 607    plotting.plot_rolling_returns(
 608        returns,
 609        factor_returns=benchmark_rets,
 610        live_start_date=live_start_date,
 611        cone_std=None,
 612        volatility_match=(benchmark_rets is not None),
 613        legend_loc=None,
 614        ax=ax_rolling_returns_vol_match,
 615    )
 616    ax_rolling_returns_vol_match.set_title(
 617        "Cumulative returns volatility matched to benchmark"
 618    )
 619
 620    plotting.plot_rolling_returns(
 621        returns,
 622        factor_returns=benchmark_rets,
 623        logy=True,
 624        live_start_date=live_start_date,
 625        cone_std=cone_std,
 626        ax=ax_rolling_returns_log,
 627    )
 628    ax_rolling_returns_log.set_title("Cumulative returns on logarithmic scale")
 629
 630    plotting.plot_returns(
 631        returns,
 632        live_start_date=live_start_date,
 633        ax=ax_returns,
 634    )
 635    ax_returns.set_title("Returns")
 636
 637    if benchmark_rets is not None:
 638        plotting.plot_rolling_beta(returns, benchmark_rets, ax=ax_rolling_beta)
 639
 640    plotting.plot_rolling_volatility(
 641        returns, factor_returns=benchmark_rets, ax=ax_rolling_volatility
 642    )
 643
 644    plotting.plot_rolling_sharpe(returns, ax=ax_rolling_sharpe)
 645
 646    # Drawdowns
 647    plotting.plot_drawdown_periods(returns, top=5, ax=ax_drawdown)
 648
 649    plotting.plot_drawdown_underwater(returns=returns, ax=ax_underwater)
 650
 651    plotting.plot_monthly_returns_heatmap(returns, ax=ax_monthly_heatmap)
 652    plotting.plot_annual_returns(returns, ax=ax_annual_returns)
 653    plotting.plot_monthly_returns_dist(returns, ax=ax_monthly_dist)
 654
 655    plotting.plot_return_quantiles(
 656        returns, live_start_date=live_start_date, ax=ax_return_quantiles
 657    )
 658
 659    if bootstrap and (benchmark_rets is not None):
 660        ax_bootstrap = plt.subplot(gs[i, :])
 661        plotting.plot_perf_stats(returns, benchmark_rets, ax=ax_bootstrap)
 662    elif bootstrap:
 663        raise ValueError("bootstrap requires passing of benchmark_rets.")
 664
 665    for ax in fig.axes:
 666        ax.tick_params(
 667            axis="x",
